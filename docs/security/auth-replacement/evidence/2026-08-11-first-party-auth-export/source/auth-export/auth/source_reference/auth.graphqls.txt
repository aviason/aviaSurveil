type Query {
  authSessions: [AuthSession!]!
}

type Mutation {
  authRegister(input: AuthRegisterInput!): Boolean!
  authLogin(input: AuthLoginInput!): AuthPayload!
  authRefresh(input: AuthRefreshInput!): AuthPayload!
  authLogout(input: AuthLogoutInput): Boolean!
  authLogoutAll(input: AuthLogoutAllInput): Boolean!
  authRevokeSession(sessionID: ID!): Boolean!
  authStepUp(input: AuthStepUpInput!): AuthStepUpResult!
  changePassword(input: ChangePasswordInput!): Boolean!
}

input AuthRegisterInput {
  name: String
  surname: String
  username: String
  email: String
  phone: String
  password: String!
  confirmPassword: String
  cityID: ID
}

input AuthLoginInput {
  username: String
  password: String!
  deviceID: String!
  clientID: String!
  platform: String
  appVersion: String
  userAgent: String
}

input AuthRefreshInput {
  refreshToken: String!
  deviceID: String!
  clientID: String!
}

input AuthLogoutInput {
  refreshToken: String
}

input AuthLogoutAllInput {
  includeCurrent: Boolean
}

input AuthStepUpInput {
  password: String
  currentPassword: String
}

input ChangePasswordInput {
  currentPassword: String!
  newPassword: String!
}

type AuthPayload {
  accessToken: AuthTokenEnvelope!
  refreshToken: AuthTokenEnvelope!
  user: AuthUser!
}

type AuthTokenEnvelope {
  token: String!
  expiration: String
}

type AuthUser {
  id: ID!
  username: String!
  fullName: String!
  name: String
  surname: String
  email: String
  phone: String
  city: String
  pictureURL: String
  role: String!
  roles: [String!]!
  authorityCodes: [String!]!
}

type AuthSession {
  sessionID: ID!
  current: Boolean!
  clientID: String!
  platform: String
  appVersion: String
  deviceLabel: String!
  createdAt: String!
  lastUsedAt: String!
  authTime: String!
  absoluteExpiresAt: String!
  idleExpiresAt: String!
  lastStepUpAt: String
  revokedAt: String
}

type AuthStepUpResult {
  validUntil: String!
}
